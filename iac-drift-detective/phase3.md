## 2. Step Functions State Machine

### step_functions/drift_workflow.asl.json
```json
以下の構造でASL (Amazon States Language) を定義:

{
  "Comment": "IaC Drift Detective - ドリフト検知から修復PRまでのワークフロー",
  "StartAt": "DetectDrift",
  "States": {
    "DetectDrift": {
      // drift-detector Lambda呼び出し
      // ResultPath: "$.detection_result"
      // タイムアウト: 120秒
      // Retry: IntervalSeconds=10, MaxAttempts=2, BackoffRate=2
      // Next: CheckDriftExists
    },
    "CheckDriftExists": {
      // Choice State
      // $.detection_result.drift_detected == true → AnalyzeDrift
      // false → NoDriftFound
    },
    "NoDriftFound": {
      // Succeed State
      // "ドリフトなし - 正常終了"
    },
    "AnalyzeDrift": {
      // bedrock-analyzer Lambda呼び出し
      // InputPath: "$.detection_result"
      // ResultPath: "$.analysis_result"
      // タイムアウト: 360秒
      // Retry: IntervalSeconds=30, MaxAttempts=2, BackoffRate=1.5
      // Next: CreatePR
    },
    "CreatePR": {
      // pr-creator Lambda呼び出し
      // InputPath: "$.analysis_result"
      // ResultPath: "$.pr_result"
      // タイムアウト: 60秒
      // Retry: IntervalSeconds=15, MaxAttempts=2
      // Next: WorkflowComplete
    },
    "WorkflowComplete": {
      // Succeed State
    }
  }
}
```
